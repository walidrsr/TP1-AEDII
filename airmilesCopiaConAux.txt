
Categoria ES enum {Bronce, Plata, Oro, Platino} 
Id ES Z
nombrePromocion ES String
promocion ES struct<
    factor: R
    tope: R --0 indica que no tiene tope 
    gastadas: R   
    >

vuelo ES struct <
    distancia: R
    millasGanadas: R 
    promocion: String 
    factor: R
>

transaccion ES struct<
    identificador_usuario2:Z 
    monto: Z
    >
    
usuario ES struct <
    millasTotales: R
    millasDisponibles: R
    millasCanjeadas: R 
    historialVuelos: seq<vuelo>
    historialTransacciones: seq <transaccion>
    >

    TAD AirMiles {

    obs usuarios: dicc <Id,usuario>
    obs promociones: dicc <nombrePromocion,promocion>

    crearSistema(): AirMiles // 
        requiere: {true}
        asegura: {res.usuarios = {}}
        asegura: {res.promociones = setKey({},'NoPromo', factorNoPromo() )} 
        

    //listo
    registrarUsuario(inout a: AirMiles, in id_usuario: Z):
        requiere: {a0 = a}
        requiere: {not estaRegistrado(a.usuarios, id_usuario)}
        requiere: {id_usuario > 0} 
        asegura:  {a.usuarios = setKey (a0.usuarios, id_usuario, nuevo_usuario())  }   
        asegura:  {a.promociones = a0.promociones}                                                                                      

    registrarUsuarioConReferidor(inout a: AirMiles, in nuevo_usuario: Z, in id_referidor: Z)
        requiere: {a0 = a} 
        requiere: {(not estaRegistrado(a.usuarios, nuevo_usuario) and estaRegistrado(a.usuarios, referidor))}
        requiere: {nuevo_usuario > 0}     
        asegura:  {a.usuarios = setKey(setKey(a0.usuarios, id_referidor, actualizarReferidor(a0.usuarios[id_referidor])) ,nuevo_usuario,nuevoUsuarioReferido())} 
        asegura:  {a.promociones = a0.promociones}   

    acumularMillas(inout a: AirMiles, in id_usuario: Z, in distancia: R, in nombrePromo: String)
        requiere: {a0 = a}
        requiere: {estaRegistrado(a.usuarios, id_usuario)}
        requiere: {distancia > 0}
        requiere: {nombrePromo in claves(a0.promociones)} 
       
        requiere {a0.promociones[nombrePromo].tope > 0  ->L MillasGanadasMasMillasGastadas(a.usuarios[id_usuario],distancia,a.usuarios[nombrePromo] < a0.promociones[nombrePromo].tope )}        
        asegura : {a.usuario=setKey(   a0.usuario, id_usuario,actualizarAcumularMillas(a0.usuario[id_usuario],distancia,a0.promocion[nombrePromo],nombrePromo)    )}
        
        
        //ver asegura: {ifThenElse(a0.promociones.tope = MillasGanadasMasMillasGastadas(a0.usuarios[id\_usuario],distancia,a0.usuarios[nombrePromo]) Y  a0.promociones.tope > 0 , // que pasa si tope ->  0(no tiene tope) = 0(no tiene gastadas) + 0 (esPlatino) 
                a.promociones = delKey(a0.promociones,nombrePromo),     
                a.promociones = setKey(a0.promociones, nombrePromo, gastadas + millasGanadasEnVuelo(a0.usuarios[id_usuario], distancia, nombrePromo))
        )} 
    //listo
    canjearMillas(inout a: AirMiles, in id_usuario: Z, in millas: R): // chequear si está bien
        requiere: {a0 = a}
        requiere: {estaRegistrado(a.usuarios, id_usuario)}
        requiere: {a.usuarios[id_usuario].millasDisponibles >= millas}
        asegura:  {a.usuarios = setKey(a0.usuarios, id_usuario,actualizarMillasCanjeadas(a0.usuarios[id_usuario],millas))}
        asegura:  {a.promociones = a0.promociones} 


    transferirMillas(inout a: AirMiles, in id_Da: Z, in id_Recibe: Z, in millas: R): 
        requiere: {a0 = a}
        requiere: {Da != Recibe}
        requiere: {estaRegistrado(a.usuarios, Da) and estaRegistrado(a.usuarios, Recibe)}
        requiere {a.usuarios[Da].millasDisponibles >= millas}
        asegura: {a.usuarios = setKey(setKey(a0.usuarios, Da, actualizarAcumularMillasDa(a0.usuarios[id_Da],millas,id_Recibe)),Recibe,actualizarMillasTransferidasRecibe(a.usuarios[id_Recibe],millas,id_Da))
      

    //listo
    consultarSaldo(in a: AirMiles, in id_usuario: Z): R
        requiere: {estaRegistrado(a.usuarios, Z)}
        asegura: {res = a.usuarios[id_usuario].millasDisponibles}

    
    //listo
    consultarCategoria (in a: AirMiles, in id_usuario: Z): Categoria
        requiere: {estaRegistrado(a.usuarios, id_usuario)}
        asegura: {esBronce(a.usuarios[id_usuario].millasTotales) <-> res = Categoria.Bronce}
        asegura: {esPlata(a.usuarios[id_usuario].millasTotales) <-> res = Categoria.Plata}
        asegura: {esOro(a.usuarios[id_usuario].millasTotales) <-> res = Categoria.Oro} 
        asegura: {esPlatino(a.usuarios[id_usuario].millasTotales) <-> res = Categoria.Platino}

    //listo
    rankingMillas (in a: AirMiles, in cant_ids: Z): seq<Id> 
        requiere: {cant_ids > 0}
        asegura: {ifThenElse(Z > |a.usuarios|, |res| = |a.usuarios|, |res| = cant_ids) y PerteneceSeq(res,claves(a.usuario)) }
        asegura: {para todo 0 <= i < |res|-1 ->luego a.usuarios[res[i]].millasDisponibles >= a.usuarios[res[i+1]].millasDisponibles}
        asegura: {(para todo i : Id) i E a.usuarios Y i !E res ->L a.usuarios[res[|res|-1]].millasDisponibles > a.usuarios[i].millasDisponibles } //se fija que sean los mejores

    resumenTransacciones(in a: AirMiles, int id_usuario: Z): tuple(R,R,R)
        requiere: {estaRegistrado(a.usuarios, id_usuario)}
        asegura: {res = (totalMillasPorViaje(a.usuario[id_usuario].historialVuelos), a.usuario[id_usuario].millasCanjeadas,millasTransferidasGeneral(a.usuarios[id_usuario].historial_transacciones))}

    iniciarPromocion(inout a: AirMiles, in nombre: String, in factor: R):
        requiere: {a0 = a}
        requiere: {factor >= 1}
        requiere: {nombre not in claves(a0.promociones)}
        asegura:  {a.promociones = setKey(a0.promociones, nombre, (factor,0,0))}
        asegura:  {a.usuarios = a0.usuarios}

    iniciarPromocionConTope(inout a: AirMiles, in nombre: String, in factor: R, in tope: R):
        requiere: {a0 = a}
        requiere: {factor >= 1}
        requiere: {tope > 0}
        requiere: {nombre not in claves(a0.promociones)}
        asegura:  {a.promociones = setKey(a0.promociones, nombre, (factor,tope,0))}
        asegura:  {a.usuarios = a0.usuarios}

    finalizarPromocion(inout a: AirMiles, in nombre: String):
        requiere: {a0 = a}
        requiere: {nombre in claves(a0.promociones) and nombre != "NoPromo"}
        asegura:  {a.promociones = delKey(a0,nombre)}
        asegura:  {a.usuarios = a0.usuarios}

    elQueTieneMasMillas(in a: AirMiles): Id
        requiere: {|a.usuarios| > 0}
        asegura:  {(para toda key de tipo id), key in claves(a.usuarios) ->luego a.usuarios[res].millasTotales >= a.usuarios[key].millasTotales}
        
    elMasCanjeador(in a: AirMiles): Id
        requiere: {|a.usuarios| > 0}
        asegura: {(para toda key de tipo id), key in claves(a.usuarios) ->luego (millasCanjeadasMillasTransferidas(a.usuarios,res)) >= (millasCanjeadasMillasTransferidas(a.usuarios,key))}
    
    paresTransferidoresExclusivos(in a: AirMiles): seq<struct(Id:Z,Id:Z)>
        requiere: {true}
        asegura: {NoHayTuplasRepetidas(res) yL transferenciasExclusivas(a.usuarios,res)  }
    
    /**************************** AUXILIARES y PREDS ****************************/


    aux factorCategoria(u:usuario): R = ifThenElse(u.millasTotales<10000, 1 ,ifThenElse(u.millasTotales <50000, 1.5 ,ifThenElse(u.millasTotales<1000000, 2 ,0)))

    aux nuevoUsuario(): usuario = (millasTotales: 0 , millasDisponibles : 0, millasCanjeadas: 0,historialVuelos: <>, historialTransacciones: <>)
    aux factorNoPromo():promocion = (factor: 1,tope: 0,gastadas:0)
    
    aux bonoReferidor (u:usuario) : R =ifThenElse(esPlatino(u),0.0,500.0)
    
    aux actualizarReferidor (u:usuario) :usuario =  (
        millasTotales : u.millasTotales + bonoReferidor(u)
        millasDisponibles : u.millasDisponibles + bonoReferidor(u)
        millasCanjeadas : u.millasCanjeadas
        historialVuelos : u.historialVuelos
        historialTransacciones : u.historialTransacciones
    )

    aux nuevoUsuarioReferido(): usuario = (millasTotales: 500 , millasDisponibles : 500, millasCanjeadas: 0, historialVuelos: <>, historialTransacciones: <>)

    aux millasGanadasEnVuelo(u:usuario, distancia: R, p: promocion): R = factorCategoria(u) * distancia * p.factor
    
    aux actualizarAcumularMillas(u:usuario, distancia: R, p: promocion, nombrePromo:String): usuario (
        millasTotales: u.millasTotales + millasGanadasEnVuelo(u,distancia,p)
        millasDisponibles: u.millasDisponibles + millasGanadasEnVuelo(u,distancia,p)
        millasCanjeadas : u.millasCanjeadas
        historialVuelos : u.historialVuelos ++ <(distancia,millasGanadasEnVuelo(u,distancia,p),nombrePromo,p.factor)>
        historialTransacciones: u.historialTransacciones    
    )

    aux MillasGanadasMasMillasGastadas (u:usuario, distancia:R,p:promocion): R = millasGanadasEnVuelo(u,distancia,p) + p.gastadas
    
    aux actualizarMillasCanjeadas (u:usuarios,millas:R) : usuario (
        millasTotales: u.millasTotales
        millasDisponibles : u.millasDisponibles - millas
        millasCanjeadas : u.millasCanjeadas + millas
        historialVuelos:u.historialVuelos
        historialTransacciones: u.historialTransacciones
    )
    
    aux actualizarAcumularMillasDa (u:usuario,millas:R,id_Recibe:z) : usuario=(
        millasTotales: u.millasTotales
        millasDisponibles: u.millasDisponibles - millas 
        millasCanjeadas: u.millasCanjeadas
        historialVuelos: u.historialVuelos
        historialTransacciones : u.historialTransacciones ++ <(id_Recibe,millas * -1)>
    )

    aux actualizarAcumularMillasRecibe (u:usuario,millas:R,id_Da:z) : usuario=(
        millasTotales: u.millasTotales
        millasDisponibles: u.millasDisponibles + millas 
        millasCanjeadas: u.millasCanjeadas
        historialVuelos: u.historialVuelos
        historialTransacciones : u.historialTransacciones ++ <(id_Da,millas)>
    )

    pred esBronce(millas: Z) {millas < 10000}
    pred esPlata(millas: Z) {millas >= 10000 and millas <= 49999}
    pred esOro(millas: Z) {millas >= 50000 and millas <= 999999}
    pred esPlatino(millas: Z) {millas >= 1000000}

    pred PerteneceSeq(s: seq<Id>, keys: seq<Id>) {para todo 0 <= i < |s| ->luego s[i] E keys} // E = Pertenece

    aux cantidadMillasDadas( transferencias: seq<transaccion>) : R = -1* (sumatoria de i = 0 hasta i = |v|-1 ifThenElse(transferencias[i].monto > 0,0,transferencias[i].monto))
    aux millasCanjeadasMillasTransferidas(usuarios:dicc<id,usuario>,id:Z) : R = usuarios[id].millasCanjeadas + cantidadMillasDadas(usuarios[id].historialTransacciones)
    
    aux millasTransferidasGeneral( t seq<transaccion>): R = sumatoria de i = 0 hasta i = |t|-1 de t[i].monto
    aux totalMillasPorViaje( v seq<vuelos>): R = sumatoria de i = 0 hasta i = |v|-1 de v[i].millasGanadas

    pred NoHayTuplasRepetidas(s:seq<struct(id1:Z,id2:Z)>){
        (vi:z)(0<=i<|s| ->luego not (Existe j:z)(0<=j<|s| y j/=i ->L ((s[i].id1=s[j].id1 o s[i].id1 = s[j].id2) Y (s[i].id2=s[j].id1 o s[i].id2=s[j].id2) )
    }

    pred transferenciasExclusivas(a: AirMiles, s: seq<struct(id1:Z,id2:Z)>){
        para toda tupla in s -> (para toda transaccion in a.usuarios[id1].historial_transacciones -> transaccion.identificador_usuario2 = id2)
        Y
        para toda tupla in s -> (para toda transaccion in a.usuarios[id2].historial_transacciones -> transaccion.identificador_usuario2 = id1)
    }
    
    pred estaRegistrado(d:dicc<id,usuario>,id_usuario:Z) {id_usuario pertenece d }
}}
