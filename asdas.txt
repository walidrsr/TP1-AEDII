TAD AirMiles {

    Categoria ES enum {Bronce, Plata, Oro, Platino}  //CreoQueNoHaceFAlta
    Id ES Z
    nombrePromocion ES String
    promocion ES struct<
        factor: R
        tope: R --0 indica que no tiene tope 
        gastadas: R   
    >

    vuelo ES struct <
        distancia: R
        millasGanadas: R 
        promocion: String 
        factor: R
    >

    transaccion ES struct<
        identificador_usuario2:Z 
        monto: Z
    >
    
    usuario ES struct <
        millasTotales: R
        millasDisponibles: R
        millasCanjeadas: R 
        historialVuelos: seq<vuelo>
        historialTransacciones: seq <transaccion>
    >

    


    obs usuarios: dicc <Id,usuario>
    obs promociones: dicc <nombrePromocion,promocion>


    pred estaRegistrado(d:dicc<id,usuario>,id_usuario:Z) {id_usuario pertenece d }
    aux factorCategoria(u:usuario): R = ifThenElse(u.millasTotales<10000, 1 ,ifThenElse(u.millasTotales <50000, 3/2 ,ifThenElse(u.millasTotales<1000000, 2 ,0)))


    crearSistema(): AirMiles // 
        requiere: {true}
        asegura: {res.usuarios = {}}
        asegura: {res.promociones = setKey({},'NoPromo', (1,0,0) )} // (factor, tope, gastadas)
        

    //listo
    registrarUsuario(inout a: AirMiles, in id_usuario: Z): // noo, solo en este proc y en el regristrarUsuarioConReferidor
        requiere: {a0 = a}
        requiere: {not estaRegistrado(a.usuarios, id_usuario)}
        requiere: {id_usuario > 0} 
        asegura:  {a.usuarios = setKey (a0.usuarios, id_usuario,(0,0,0,<>,<>))  }   
        asegura:  {a.promociones = a0.promociones}       // (totales,disp,canjeadas,histvuelos,histTransacc)                                                                                  


   
    registrarUsuarioConReferidor(inout a: AirMiles, in nuevo_usuario: Z, in referidor: Z)
        requiere: {a0 = a} 
        requiere: {(not estaRegistrado(a.usuarios, nuevo_usuario) and estaRegistrado(a.usuarios, referidor))}
        requiere: {nuevo_usuario > 0}     
        asegura:  { esPlatino(a0.usuarios[referidor].millasTotales )        ->L a.usuarios = setKey(setKey(a0.usuarios, id_referidor, (a0.usuario[id\_referidor].millasTotales, a0.usuario[id_referidor].millasDisponibles + 500, a0.usuario[id_referidor].historialVuelos, a0.usuario[id\_referidor].historialTransacciones),nuevo_usuario,(500,500,0,<>,<>)))} 
        asegura:  { not esPlatino(a0.usuarios[referidor].millasTotales)     ->L a.usuarios = setKey(setKey(a0.usuarios, id\_referidor, (a0.usuario[id\_referidor].millasTotales  + 500, a0.usuario[id\_referidor].millasDisponibles + 500, a0.usuario[id\_referidor].historialVuelos, a0.usuario[id\_referidor].historialTransacciones),nuevo_usuario,(500,500,0,<>,<>))}
        asegura:  {a.usuario = setKey(a1.usuario,nuevo_usuario,(500,500,0,<>,<>))}  //como en cualquiera de los casos modificamos a1, aca va a1, no a0
        asegura:  {a.promociones = a0.promociones}   //aca da igual porque ni tocamos promociones    



    acumularMillas(inout a: AirMiles, in id_usuario: Z, in distancia: R, in nombrePromo: String)
        requiere: {a0 = a}
        requiere: {estaRegistrado(a.usuarios, id_usuario)}
        requiere: {distancia > 0}
        requiere: {nombrePromo in claves(a0.promociones)} 
        requiere : {(a0.promociones[nombrePromo].tope > 0 ->L (millasGanadasEnVuelo(a0, id_usuario, distancia, nombrePromo) + a0.promociones[nombrePromo].gastadas)  < a0.promociones[nombrePromo].tope)} 
        asegura : {a.usuario=setKey(a0.usuario,id_usuario,(a0.usuario[id_usuario].millasTotales , a0.usuario[id_usuario].millasDisponibles + millasGanadasEnVuelo(a0,id_usuario,distancia,nombrePromo) , a0.usuarios[id_usuario].millasCanjeadas, a0.usuarios[id_usuario].historialVuelos ++ <(distancia,millasGanadasEnVuelo(a0,id_usuario,distancia,nombrePromo),nombrePromo,a0.promociones[nombrePromo].factor)>, a0.usuarios[id_usuario].historial_transacciones))}
        asegura: {ifThenElse(a0.promociones.tope = a0.promociones.gastadas + millasGanadasEnVuelo(a0, id_usuario, distancia, nombrePromo), a.promociones = delKey(a0.promociones,nombrePromo), a.promociones = setKey(a0.promociones, nombrePromo, gastadas + millasGanadasEnVuelo(a0, id_usuario, distancia, nombrePromo)))} //ver a0.promociones= :::

    aux millasGanadasEnVuelo(a: AirMiles, id: Z, distancia: R, nombrePromo: String): R = factorCategoria(a.usuario[Id]) * distancia * a.promociones[nombrePromo].factor

    //listo
    canjearMillas(inout a: AirMiles, in id_usuario: Z, in millas: R): // chequear si está bien
        requiere: {a0 = a}
        requiere: {estaRegistrado(a.usuarios, id_usuario)}
        requiere: {a.usuarios[id_usuario].millasDisponibles >= millas}
        asegura:  {a.usuarios = setKey(a0.usuarios, id_usuario, (a0.usuario[id_usuario].totales, a0.usuario[id_usuario].millasDisponibles - millas, a0.usuario[id_usuario].millasCanjeadas + millas, a0.usuario[referidor].historialVuelos,a0.usuario[referidor].historialTransacciones))}
        asegura:  {a.promociones = a0.promociones} 

    //categoria,metavariable en asegura?
    transferirMillas(inout a: AirMiles, in Da: Z, in Recibe: Z, in millas: R): // chequear si está bien
        requiere: {a0 = a, a1 = a}
        requiere: {Da != Recibe}
        requiere: {estaRegistrado(a.usuarios, Da) and estaRegistrado(a.usuarios, Recibe)}
        requiere {a0.usuarios[Da].millasDisponibles >= millas}
        asegura: {a1.usuarios = setKey(a0.usuarios, Da, (a0.usuario[Da].totales, a0.usuario[Da].millasDisponibles - millas, a0.usuario[Da].millasCanjeadas, a0.usuario[Da].historialVuelos, a0.usuario[Da].historial_transacciones ++ <(Recibe,monto*-1)>))} //monto*-1 no es millas?
      FALTA EL QUE Recibe  //asegura: {a.usuarios = setKey(a0.usuarios, Da, (a0.usuario[Da].totales, a0.usuario[Da].millasDisponibles - millas, a0.usuario[Da].millasCanjeadas, a0.usuario[Da].historialVuelos, a0.usuario[Da].historial_transacciones ++ <(Recibe,monto*-1)>))} //en los ultimos 2 asegura estabamos igualando a a un diccionario, cuando a es el tad entero. Complete con a1.usuarios y a.usuarios respectivamente.
        asegura: {a.promociones = a0.promociones} 

    //listo
    consultarSaldo(in a: AirMiles, in id_usuario: Z): R
        requiere: {estaRegistrado(a.usuarios, Z)}
        asegura: {res = a.usuarios[id_usuario].millasDisponibles}

    ¿¿¿¿<->???
    consultarCategoria (in a: AirMiles, in id_usuario: Z): Categoria
        requiere: {estaRegistrado(a.usuarios, id_usuario)}
        asegura: {esBronce(a.usuarios[id_usuario].millasTotales) <-> res = Categoria.Bronce}
        asegura: {esPlata(a.usuarios[id_usuario].millasTotales) <-> res = Categoria.Plata}
        asegura: {esOro(a.usuarios[id_usuario].millasTotales) <-> res = Categoria.Oro} 
        asegura: {esPlatino(a.usuarios[id_usuario].millasTotales) <-> res = Categoria.Platino}


    pred esBronce(millas: Z) {millas < 10000}
    pred esPlata(millas: Z) {millas >= 10000 and millas <= 49999}
    pred esOro(millas: Z) {millas >= 50000 and millas <= 999999}
    pred esPlatino(millas: Z) {millas >= 1000000}

    //listo
    rankingMillas (in a: AirMiles, in cant_ids: Z): seq<Id> // problema con rango de i
        requiere: {cant_ids > 0}
        asegura: {ifThenElse(Z > |a.usuarios|, |res| = |a.usuarios|, |res| = cant_ids) y PerteneceSeq(res,claves(a.usuario)) }
        asegura: {para todo 0 <= i < |res|-1 ->luego a.usuarios[res[i]].millasDisponibles >= a.usuarios[res[i+1]].millasDisponibles}
        asegura: {(para todo i : Id) i E a.usuarios Y i !E res ->L res[|res|-1] > a.usuarios[i](esto es un struct)} //se fija que sean los mejores


    pred PerteneceSeq(s: seq<Id>, keys: seq<Id>) {para todo 0 <= i < |s| ->luego s[i] E keys} // E = Pertenece

    resumenTransacciones(in a: AirMiles, int id_usuario: Z): tuple(R,R,R)
        requiere: {estaRegistrado(a.usuarios, id_usuario)}
        asegura: {res = (totalMillasPorViaje(a.usuario[id_usuario].historialVuelos), a.usuario[id_usuario].millasCanjeadas,millasTransferidasGeneral(a.usuarios[id_usuario].historial_transacciones))}

    aux millasTransferidasGeneral(in t seq<transaccion>): R = sumatoria de i = 0 hasta i = |t|-1 de t[i].monto
    aux totalMillasPorViaje(in v seq<vuelos>): R = sumatoria de i = 0 hasta i = |v|-1 de v[i].millasGanadas

    iniciarPromocion(inout a: AirMiles, in nombre: String, in factor: R):
        requiere: {a0 = a}
        requiere: {factor >= 1}
        requiere: {nombre not in claves(a0.promociones)}
        asegura:  {a.promociones = setKey(a0.promociones, nombre, (factor,0,0))}
        asegura:  {a.usuarios = a0.usuarios}

    iniciarPromocionConTope(inout a: AirMiles, in nombre: String, in factor: R, in tope: R):
        requiere: {a0 = a}
        requiere: {factor >= 1}
        requiere: {tope > 0}
        requiere: {nombre not in claves(a0.promociones)}
        asegura:  {a.promociones = setKey(a0.promociones, nombre, (factor,tope,0))}
        asegura:  {a.usuarios = a0.usuarios}

    finalizarPromocion(inout a: AirMiles, in nombre: String):
        requiere: {a0 = a}
        requiere: {nombre in claves(a0.promociones) and nombre != "NoPromo"}
        asegura:  {a.promociones = delKey(a0,nombre)}
        asegura:  {a.usuarios = a0.usuarios}

    elQueTieneMasMillas(in a: AirMiles): Id
        requiere: {|a.usuarios| > 0}
        asegura:  {(para toda key de tipo id), key in claves(a.usuarios) ->luego a.usuarios[res].millasTotales >= a.usuarios[key].millasTotales}
        asegura :    { no existe otro usuario que tenga mas millas que res ??}

    
    ///ver
    elMasCanjeador(in a: AirMiles): Id
        requiere: {|a.usuarios| > 0}
        asegura: {(para toda key de tipo id), key in claves(a.usuarios) ->luego (a.usuarios[res].millasCanjeadas + cantidadMillasDadas(a.usuarios[res].historial_transacciones)) >= (a.usuarios[key].millasCanjeadas + cantidadMillasDadas(a.usuarios[key].historial_transacciones))}

    //ver
    elMasCanjeador(in a: AirMiles): Id
        requiere: {|a.usuarios| > 0}
        asegura: {para toda key in claves(a.usuarios) ->luego (a.usuarios[res].millasTotales - a.usuarios[res].millasDisponibles) >= (a.usuarios[key].millasTotales - a.usuarios[key].millasDisponibles)}


    aux cantidadMillasDadas(in transferencias: seq<transaccion>) : R = -1* (sumatoria de i = 0 hasta i = |v|-1 ifThenElse(transferencias[i].monto > 0,0,transferencias[i].monto))
    
    paresTransferidoresExclusivos(in a: AirMiles): seq<struct(Id:Z,Id:Z)>
        requiere: {true}
        asegura: {NoHayTuplasRepetidas(res) yL transferenciasExclusivas(a.usuarios,res)  } // no hace falta aclarar que existe almenos una transferencia??
    
    pred NoHayTuplasRepetidas(s:seq<struct(id1:Z,id2:Z)>){
        (vi:z)(0<=i<|s| ->luego not (Existe j:z)(0<=j<|s| y j/=i ->L ((s[i].id1=s[j].id1 o s[i].id1 = s[j].id2) Y (s[i].id2=s[j].id1 o s[i].id2=s[j].id2) )
    }

    pred transferenciasExclusivas(a: AirMiles, s: seq<struct(id1:Z,id2:Z)>){
        para toda tupla in s -> (para toda transaccion in a.usuarios[id1].historial_transacciones -> transaccion.identificador_usuario2 = id2)
        Y
        para toda tupla in s -> (para toda transaccion in a.usuarios[id2].historial_transacciones -> transaccion.identificador_usuario2 = id1)
    }
}
